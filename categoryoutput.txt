CREATE TABLE Category (
      Category_ID NUMBER PRIMARY KEY,
      Category_Name VARCHAR2(50) NOT NULL UNIQUE
  );

SELECT
       c.Category_Name,
       p.Product_ID,
       p.Product_Name,
       p.Brand,
       p.Price,
       p.Stock
    FROM Category c
    JOIN Product p
   ON c.Category_ID = p.Category_ID
   ORDER BY c.Category_Name;
